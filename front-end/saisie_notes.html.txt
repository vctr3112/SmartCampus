<script>
document.addEventListener("DOMContentLoaded", function() {

var selectCours = document.getElementById("select-cours");
var selectClasse = document.getElementById("select-classe");
var dateExamen = document.getElementById("date-examen");

//chargement des cours de l'enseignant
fetch("traitement_saisie_notes.php?cours")
.then(function(response) {
return response.json();
})
.then(function(data) {
data.forEach(function(cours) {
var option = document.createElement("option");
option.value = cours.id_cours;
option.textContent = cours.nom_cours;
selectCours.appendChild(option);
});
});

//chargement des classes quand on change de cours
selectCours.addEventListener("change", function() {
var id_cours = this.value;
selectClasse.innerHTML = '<option value="">-- Sélectionner une classe --</option>';
document.getElementById("liste-etudiants").innerHTML = "";

if (id_cours) {
fetch("traitement_saisie_notes.php?classes&id_cours=" + id_cours)
.then(function(response) {
return response.json();
})
.then(function(data) {
data.forEach(function(classe) {
var option = document.createElement("option");
option.value = classe.filiere + "|" + classe.niveau;
option.textContent = classe.filiere + " - " + classe.niveau;
selectClasse.appendChild(option);
});
});
}
});

//chargement des étudiants quand on change de classe
selectClasse.addEventListener("change", function() {
var id_cours = selectCours.value;
var classeVal = this.value;
document.getElementById("liste-etudiants").innerHTML = "";

if (id_cours && classeVal) {
var parts = classeVal.split("|");
var filiere = parts[0];
var niveau = parts[1];

fetch("traitement_saisie_notes.php?etudiants&id_cours=" + id_cours + "&filiere=" + encodeURIComponent(filiere) + "&niveau=" + encodeURIComponent(niveau))
.then(function(response) {
return response.json();
})
.then(function(data) {
var tbody = document.getElementById("liste-etudiants");
data.forEach(function(etudiant) {
var tr = document.createElement("tr");
tr.innerHTML =
"<td>" + etudiant.nom + "</td>" +
"<td>" + etudiant.prenom + "</td>" +
"<td><input type='number' name='notes[" + etudiant.id_etudiant + "]' min='0' max='20' step='0.25'></td>";
tbody.appendChild(tr);
});

//mise à jour des champs cachés
document.getElementById("hidden-cours").value = id_cours;
});
}
});

//mise à jour du champ caché date
dateExamen.addEventListener("change", function() {
document.getElementById("hidden-date").value = this.value;
});

});
</script>
