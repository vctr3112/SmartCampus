<script>

document.addEventListener("DOMContentLoaded", function() {
 
fetch("traitement_dashboard_enseignant.php")
.then(function(response) {
return response.json();
})
.then(function(data) {

//affichage du prochain cours
var edtContent = document.getElementById("edt-content");
if (data.prochain_cours) {
edtContent.innerHTML =
"<p><strong>" + data.prochain_cours.nom_cours + "</strong></p>" +
"<p>Date : " + data.prochain_cours.date + "</p>" +
"<p>Horaire : " + data.prochain_cours.heure_debut + " - " + data.prochain_cours.heure_fin + "</p>" +
"<p>Salle : " + data.prochain_cours.salle + "</p>";
} else {
edtContent.innerHTML = "<p>Aucun cours à venir.</p>";
}

//affichage des notes à saisir
var notesContent = document.getElementById("notes-content");
if (data.notes && data.notes.length > 0) {
var html = "";
data.notes.forEach(function(cours) {
var restant = cours.nb_etudiants - cours.nb_notes_saisies;
if (restant > 0) {
html += "<p>" + cours.nom_cours + " : " + restant + " note(s) à saisir</p>";
}
});
notesContent.innerHTML = html || "<p>Toutes les notes sont saisies.</p>";
} else {
notesContent.innerHTML = "<p>Aucun cours trouvé.</p>";
}

});
 
});

</script>
